## （参考）照合に使ったスクリプト
TSVに行が増えたときも、表を作り直してから同じように数え直せます。使い方：`python3 -I count_check.py <TSVのパス> <このファイルのパス>`

```python
"""01 の対応表（出来上がったファイル）と TSV を突き合わせて件数を数える。
使い方: python3 -I count_check.py <TSV> <01のファイル>
"""
import csv, sys, re
from collections import Counter

tsv_path, md_path = sys.argv[1], sys.argv[2]

# ダイジェスト（もち作成）の集計
DIGEST_PARTNER = {"Perplexity": 27, "Claude Code": 19, "Claude": 4, "Google Gemini": 3}
DIGEST_MODEL = {"Fable": 12, "Sol": 11, "Opus": 6, "GPT-6.1 Sol": 4, "Sol (GPT-6.1 Sol)": 3,
                "Neon": 3, "GPT-6.2 Sol": 3, "Deep Research（デフォルトモデル）": 3, "Pro検索（Sol）": 2,
                "Opus（Claude Code）": 2, "Opus (Claude Code, effort High)": 2,
                "Opus (effort High)": 1, "GPT-6.2 Sol（思考ON）": 1}

with open(tsv_path, encoding="utf-8", newline="") as f:
    tsv = list(csv.DictReader(f, delimiter="\t"))
tsv_by_id = {int(r["id"]): r for r in tsv}

def plain(s):
    return s.replace("**", "").strip()

table = []
in_section = False
for line in open(md_path, encoding="utf-8"):
    if line.startswith("## "):
        in_section = line.startswith("## 対応表（全53件）")  # この節の表だけを数える
        continue
    if in_section and re.match(r"^\| \d+ \|", line):
        c = [plain(x) for x in line.rstrip("\n").strip("|").split("|")]
        table.append(dict(id=int(c[0]), date=c[1], time=c[2], pb=c[4], pa=c[5], mb=c[6], ma=c[7], note=c[8]))

def show(title, cnt, expect=None):
    print(f"[{title}] 種類数 {len(cnt)}、合計 {sum(cnt.values())}")
    for k, v in sorted(cnt.items(), key=lambda kv: (-kv[1], kv[0])):
        mark = ""
        if expect is not None:
            mark = "  … ダイジェストと一致" if expect.get(k) == v else f"  … 不一致（ダイジェスト {expect.get(k)}）"
        print(f"  {k}: {v}{mark}")
    if expect is not None:
        ok = dict(cnt) == expect
        print("  → ダイジェストの集計と" + ("すべて一致" if ok else "不一致あり"))

print("== 1. 行数 ==")
print("TSVの行数（見出し行を除く）:", len(tsv))
print("01の表の行数:", len(table))
ids_t = [r["id"] for r in table]
print("表のidに重複なし:", len(ids_t) == len(set(ids_t)))
print("表のidとTSVのidが同じ集まり:", set(ids_t) == set(tsv_by_id))
mism = [r["id"] for r in table
        if (r["date"], r["pb"], r["mb"]) != (tsv_by_id[r["id"]]["date"], tsv_by_id[r["id"]]["partner"], tsv_by_id[r["id"]]["model"])]
print("表の「日付・相手（今）・モデル（今）」がTSVと食い違う行:", mism if mism else "なし")

print()
print("== 2. 正規化前（TSVから数える） ==")
show("相手", Counter(r["partner"] for r in tsv), DIGEST_PARTNER)
show("モデル", Counter(r["model"] for r in tsv), DIGEST_MODEL)

print()
print("== 3. 正規化後（01の表から数える） ==")
prov = [r["id"] for r in table if "要確認" in r["pa"]]
show("相手", Counter(re.sub(r"（.*?）", "", r["pa"]) for r in table))
print("  うち「仮・要確認」:", len(prov), "件 → id", prov)
show("モデル", Counter(r["ma"] for r in table))
print("メモ欄に何か入る行:", sum(1 for r in table if r["note"]), "件")

print()
print("== 4. 時刻がHH:MMでない行（TSV） ==")
hhmm = re.compile(r"^([01][0-9]|2[0-3]):[0-5][0-9]$")
for r in tsv:
    if not hhmm.match(r["time"]):
        print(f"  #{r['id']} {r['date']} 時刻「{r['time'] or '（空欄）'}」")

print()
print("== 5. 重複の点検（正規化後の相手名で） ==")
pa_by_id = {r["id"]: re.sub(r"（.*?）", "", r["pa"]) for r in table}
k3 = Counter((r["date"], r["title"], pa_by_id[int(r["id"])]) for r in tsv)
print("日付・題名・相手の3点がすべて同じ組:", [k for k, v in k3.items() if v > 1] or "なし")
k2 = {}
for r in tsv:
    k2.setdefault((r["date"], r["title"]), []).append(int(r["id"]))
for k, v in k2.items():
    if len(v) > 1:
        print("日付・題名だけ同じ組（相手が違うので重複ではない）:",
              ", ".join(f"#{i}（{pa_by_id[i]}・{tsv_by_id[i]['time']}）" for i in v))
```
